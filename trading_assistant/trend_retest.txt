f_drawPlan(string id, int side, float entry, float stop, float target, float plannedRisk) =>
    int startTime = time_close
    int endTime = startTime + drawingBars * 3600000
    box rewardBox = box.new(left = startTime, top = math.max(entry, target), right = endTime, bottom = math.min(entry, target), xloc = xloc.bar_time, border_color = color.new(buyColor, 40), bgcolor = color.new(buyColor, 92))
    box riskBox = box.new(left = startTime, top = math.max(entry, stop), right = endTime, bottom = math.min(entry, stop), xloc = xloc.bar_time, border_color = color.new(sellColor, 40), bgcolor = color.new(sellColor, 92))
    line entryLine = line.new(startTime, entry, endTime, entry, xloc = xloc.bar_time, color = entryColor, width = 2)
    line stopLine = line.new(startTime, stop, endTime, stop, xloc = xloc.bar_time, color = sellColor, style = line.style_dashed)
    line targetLine = line.new(startTime, target, endTime, target, xloc = xloc.bar_time, color = buyColor, style = line.style_dashed)
    label entryTag = label.new(endTime, entry, (side == 1 ? "LONG" : "SHORT") + " · ОЖИДАНИЕ\n" + str.tostring(entry, format.mintick), xloc = xloc.bar_time, style = label.style_label_left, color = entryColor, textcolor = color.white, size = size.small)
    label stopTag = label.new(endTime, stop, "SL\n" + str.tostring(stop, format.mintick), xloc = xloc.bar_time, style = label.style_label_left, color = sellColor, textcolor = color.white, size = size.small)
    label targetTag = label.new(endTime, target, "TP · " + str.tostring(rewardRisk, "#.##") + "R\n" + str.tostring(target, format.mintick), xloc = xloc.bar_time, style = label.style_label_left, color = buyColor, textcolor = color.white, size = size.small)
    label.set_tooltip(entryTag, "Вход — стоп-ордер в модели TradingView со следующей свечи. Плановый риск до издержек: " + str.tostring(plannedRisk, "#.###") + "%. Из-за лимита стоимости риск может быть меньше настройки.")
    PlanDrawing.new(rewardBox, riskBox, entryLine, stopLine, targetLine, entryTag, stopTag, targetTag, id, startTime)

f_markDrawing(PlanDrawing drawing, int phase, int side, float entry, int endedTime) =>
    string status = phase == 2 ? "ОЖИДАНИЕ" : phase == 3 ? "АКТИВЕН В МОДЕЛИ" : phase == 4 ? "ОТМЕНЁН" : "ЗАКРЫТ В МОДЕЛИ"
    label.set_text(drawing.entryTag, (side == 1 ? "LONG" : "SHORT") + " · " + status + "\n" + str.tostring(entry, format.mintick))
    label.set_color(drawing.entryTag, phase == 4 ? color.gray : entryColor)
    box.set_bgcolor(drawing.reward, color.new(buyColor, phase == 3 ? 84 : phase == 4 ? 97 : 92))
    box.set_bgcolor(drawing.risk, color.new(sellColor, phase == 3 ? 86 : phase == 4 ? 97 : 92))
    if phase >= 2
        int endTime = phase == 4 or phase == 5 ? math.max(endedTime, drawing.createdTime + 3600000) : math.max(drawing.createdTime + drawingBars * 3600000, time_close + 3600000)
        box.set_right(drawing.reward, endTime)
        box.set_right(drawing.risk, endTime)
        line.set_x2(drawing.entryLine, endTime)
        line.set_x2(drawing.stopLine, endTime)
        line.set_x2(drawing.targetLine, endTime)
        label.set_x(drawing.entryTag, endTime)
        label.set_x(drawing.stopTag, endTime)
        label.set_x(drawing.targetTag, endTime)

// Сначала предыдущая, затем текущая 4H-свеча. Незакрытую снаружи не принимаем.
f_fourHourPacket() =>
    float fast = ta.ema(close, fastLength)
    float slow = ta.ema(close, slowLength)
    float ph = ta.pivothigh(high, obstacleStrength, obstacleStrength)
    float pl = ta.pivotlow(low, obstacleStrength, obstacleStrength)
    [time_close[1], close[1], fast[1], slow[1], fast[2], ph[1], pl[1], bar_index[1], time_close, close, fast, slow, fast[1], ph, pl, bar_index]

f_jsonText(string value) =>
    string escaped = str.replace_all(value, "\\", "\\\\")
    escaped := str.replace_all(escaped, "\"", "\\\"")
    escaped := str.replace_all(escaped, "\n", "\\n")
    escaped := str.replace_all(escaped, "\r", "\\r")
    escaped := str.replace_all(escaped, "\t", "\\t")
    "\"" + escaped + "\""

f_number(float value) =>
    na(value) ? "null" : str.tostring(value, "#.##########")

f_integer(int value) =>
    na(value) ? "null" : str.tostring(value)

f_stage(int phase) =>
    phase == 1 ? "waiting_retest" : phase == 2 ? "armed" : phase == 3 ? "active" : phase == 4 ? "canceled" : phase == 5 ? "closed" : "idle"

f_trend(int trend) =>
    trend == 1 ? "bullish" : trend == -1 ? "bearish" : "neutral"

f_emit(string kind, string id, int sequence, int phase, int side, int trend, float entry, float stop, float target, float plannedRisk, int expiresAt, string reason, int contextTime) =>
    int nextSequence = sequence + 1
    if enableAlerts and barstate.isrealtime
        string eventId = kind == "context" ? syminfo.tickerid + "|" + configId + "|context|" + str.tostring(time_close) + (barstate.isconfirmed ? "|close" : "|bootstrap|" + str.tostring(timenow)) : id + "|" + kind
        string direction = side == 1 ? "long" : side == -1 ? "short" : "none"
        string eventTitle = kind == "setup" ? "Сценарий готов" : kind == "trigger" ? "Условие входа выполнено в модели" : kind == "cancel" ? "Сценарий отменён" : kind == "exit" ? "Сценарий завершён в модели" : kind == "breakout" ? "Пробой подтверждён, ждём ретест" : "Обновление тренда 4H"
        string alertMessage = syminfo.tickerid + " · " + eventTitle + "\nТренд: " + (trend == 1 ? "бычий" : trend == -1 ? "медвежий" : "нейтральный") + "\n" + reason
        if not na(entry)
            alertMessage := alertMessage + "\n" + str.upper(direction) + " · Вход " + str.tostring(entry, format.mintick) + " · SL " + str.tostring(stop, format.mintick) + " · TP " + str.tostring(target, format.mintick) + " · " + str.tostring(rewardRisk, "#.##") + "R"
        string payload = "{\"schema_version\":1,\"strategy\":\"trend_retest_v1\",\"secret\":" + f_jsonText(webhookSecret) + ",\"event_id\":" + f_jsonText(eventId) + ",\"setup_id\":" + f_jsonText(id) + ",\"sequence\":" + str.tostring(nextSequence) + ",\"event\":" + f_jsonText(kind) + ",\"stage\":" + f_jsonText(f_stage(phase)) + ",\"symbol\":" + f_jsonText(syminfo.tickerid) + ",\"timeframe\":\"60\",\"direction\":" + f_jsonText(direction) + ",\"trend\":" + f_jsonText(f_trend(trend)) + ",\"event_time\":" + str.tostring(timenow) + ",\"bar_time\":" + str.tostring(time) + ",\"entry\":" + f_number(entry) + ",\"stop\":" + f_number(stop) + ",\"target\":" + f_number(target) + ",\"rr\":" + f_number(na(entry) ? na : rewardRisk) + ",\"risk_pct\":" + f_number(plannedRisk) + ",\"expires_at\":" + f_integer(expiresAt) + ",\"reason\":" + f_jsonText(reason) + "}"
        alert(deliveryMode == "TradingView" ? alertMessage : payload, alert.freq_all)
    nextSequence

// Линейный USD/USDT-инструмент; стоимость ограничена доступным капиталом.
f_quantity(float entry, float stop, float equity) =>
    float pointValue = syminfo.pointvalue
    float step = syminfo.mincontract
    float risk = math.abs(entry - stop)
    float quantity = 0.0
    if equity > 0 and entry > 0 and risk > 0 and pointValue > 0 and step > 0
        float riskQuantity = equity * riskPercent / 100.0 / (risk * pointValue)
        float exposureQuantity = equity * exposurePercent / 100.0 * 0.99 / (entry * pointValue)
        quantity := math.floor(math.min(riskQuantity, exposureQuantity) / step) * step
    float plannedRisk = equity > 0 ? quantity * risk * pointValue / equity * 100.0 : 0.0
    [quantity, plannedRisk]

f_obstacle(array<float> levels, int side, float entry, float target) =>
    bool blocked = false
    if useObstacles and array.size(levels) > 0
        for i = 0 to array.size(levels) - 1
            float level = array.get(levels, i)
            blocked := blocked or (side == 1 ? level > entry and level < target : level < entry and level > target)
    blocked

        array.push(breakoutTags, latestBreakTag)
        eventSequence := f_emit("breakout", setupId, eventSequence, phase, side, trend, planEntry, planStop, planTarget, planRisk, expiresAt, statusReason, contextTime)

    // Тихий часовой снимок нужен ассистенту даже при неизменном тренде.

    contextSent := true

if barstate.isrealtime and not bootstrapped
    bootstrapped := true
    if not contextSent
        string bootstrapId = setupId == "" ? syminfo.tickerid + "|context|" + str.tostring(time_close) : setupId
        eventSequence := f_emit("context", bootstrapId, eventSequence, phase, side, trend, planEntry, planStop, planTarget, planRisk, expiresAt, "Первый live-снимок. " + statusReason, contextTime)
